\begin{frame}[t]{BaLi: An Introduction}
  \visible<1->{
  What we do at BaLi
  \begin{itemize}
    \item Co-trapping \Ba{} and \Li{}.
    \item Optical trapping of ions
    \item Looking for interactions in the ultracold regime
    \end{itemize}
  }
  \visible<2->{
  Why \BaLi?
  \begin{itemize}
    \item<1-> Large mass ratio means lower collision energy
    \item<1-> Combination is chemically stable in their ground states
    \item<1-> Candidates for all-optical trapping
    \end{itemize}
  }

\end{frame}

\begin{frame}[t]{Feshbach Resonances}
    Why are we interested in Feshbach resonances?
    \begin{itemize}
      % \item<1-> Demonstration of ultracold temperatures
      \item<2-> Tuning of interactions. Can be made:
      \begin{itemize}
        \item strong ($a\rightarrow \pm\infty$)
        \item attractive ($a>0$)
        \item repulsive ($a<0$)
        \item switched off ($a=0$)
      \end{itemize}
      \item<3-> Insight into molecular spectra
      \item<4-> Formation of cold molecules in chosen states
      \end{itemize}
\end{frame}
