            E_{\mathrm{coll}}  = \frac{\mu}{m_{\mathrm{Li}}} E_{\mathrm{Li}} + \frac{\mu}{m_{\mathrm{\ba}}} \left(E_{\ba} + E_{\mathrm{mm}} \right)\label{eq:E}
        \end{equation*}

        \item[\textrightarrow] Also reduces the effect collisional heating
        \item Individual ground states also molecular ground state - low background losses
    \end{itemize}
