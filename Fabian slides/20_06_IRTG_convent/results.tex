    \visible<2>{\picpos{0.4\linewidth}{0.33\linewidth}{0.2cm}{coolscheme2}}

    \visible<4->{\picpos{0.4\linewidth}{0.33\linewidth}{0.2cm}{coolscheme0}}
    \visible<4->{\picpos{0.47\linewidth}{0.33\linewidth}{0.2cm}{coolscheme6}}
    \vspace{.28\linewidth}
    \begin{enumerate}
        \setlength\itemsep{0.5em}
    \item<1-> Prepare Doppler-cooled Ba$^+$ in $S_{\nicefrac{1}{2}}(s=\nicefrac{1}{2},
        m_s=\pm\nicefrac{1}{2})$.
    \item<2-> Prepare $\approx \SI{20}{k}$ ultracold $^6$Li atoms in hyperfine state $\ket{1}$ or $\ket{2}$.
    \item<3-> Overlap ion and atoms at magnetic field $B$ for $t_\text{int}$.
    \item<4-> Detect Ba$^+$-fluorescence on cooling transition.
    \item<5> Repeat ad infinitum.
    \end{enumerate}
\end{frame}
